#!/bin/bash

mkdir -p build_host

failed=0
total=0

for f in host_src/*.c; do
    [ -f "$f" ] || continue
    
    name=$(basename "$f" .c)
    total=$((total + 1))

    gcc -m32 -iquote include -iquote . -c "$f" -o "build_host/${name}.o" -lm

    if [ $? -ne 0 ]; then
        echo "FAIL: $f"
        failed=$((failed + 1))
    fi
done

echo "SCAN COMPLETE"
echo "Total .o files generated: $(ls -1 build_host/*.o 2>/dev/null | wc -l) / $total"

if [ $failed -eq 0 ] && [ $total -gt 0 ]; then
    echo "Exit Status: 0 (SUCCESS)"
    exit 0
else
    echo "Exit Status: 1 (FAILURE)"
    exit 1
fi
